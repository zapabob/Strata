cmake_minimum_required(VERSION 3.24)
project(spin8_prism_query_v3_test LANGUAGES C CXX)
if(NOT PRISM_SOURCE OR NOT SPIN8_SOURCE)
    message(FATAL_ERROR "Set PRISM_SOURCE and SPIN8_SOURCE to existing local sources; no download is performed")
endif()
if(NOT EXISTS "${PRISM_SOURCE}/ggml/include/ggml.h" OR NOT EXISTS "${SPIN8_SOURCE}/include/spin8/query_v3.h")
    message(FATAL_ERROR "The local pinned GGML and Spin8 v3 sources are required")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PRISM_SOURCE}" rev-parse HEAD
    RESULT_VARIABLE prism_git_result OUTPUT_VARIABLE prism_git_head
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_VARIABLE prism_git_error)
if(NOT prism_git_result EQUAL 0 OR NOT prism_git_head STREQUAL "73bfe6cc485554822f80c473a3c0c4f25e80f41b")
    message(FATAL_ERROR "PRISM_SOURCE must be the Git checkout pinned to 73bfe6cc485554822f80c473a3c0c4f25e80f41b; source exports are unsupported: ${prism_git_error}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PRISM_SOURCE}" diff --quiet HEAD -- ggml
    RESULT_VARIABLE prism_ggml_diff_result ERROR_VARIABLE prism_ggml_diff_error)
if(NOT prism_ggml_diff_result EQUAL 0)
    message(FATAL_ERROR "The pinned GGML subtree must have no tracked changes: ${prism_ggml_diff_error}")
endif()
message(STATUS "Verified Prism HEAD ${prism_git_head}; GGML subtree has no tracked changes")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_HIP OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_MUSA OFF CACHE BOOL "" FORCE)
set(GGML_CANN OFF CACHE BOOL "" FORCE)
set(GGML_WEBGPU OFF CACHE BOOL "" FORCE)
set(GGML_OPENVINO OFF CACHE BOOL "" FORCE)
set(GGML_ZDNN OFF CACHE BOOL "" FORCE)
set(GGML_VIRTGPU OFF CACHE BOOL "" FORCE)
set(GGML_VIRTGPU_BACKEND OFF CACHE BOOL "" FORCE)
set(GGML_ET OFF CACHE BOOL "" FORCE)
set(GGML_ET_SYSEMU OFF CACHE BOOL "" FORCE)
set(GGML_HEXAGON OFF CACHE BOOL "" FORCE)
set(GGML_ZENDNN OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP_FETCH OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(SPIN8_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPIN8_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
add_subdirectory("${PRISM_SOURCE}/ggml" ggml)
add_subdirectory("${SPIN8_SOURCE}" spin8)
add_executable(spin8_prism_query_v3_test ../spin8_prism_query_v3_test.cpp)
target_link_libraries(spin8_prism_query_v3_test PRIVATE spin8 ggml ggml-cpu)
target_compile_features(spin8_prism_query_v3_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(spin8_prism_query_v3_test PRIVATE /W4 /WX /utf-8 /fp:strict)
else()
    target_compile_options(spin8_prism_query_v3_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME spin8_prism_query_v3_graph COMMAND spin8_prism_query_v3_test)
