cmake_minimum_required(VERSION 3.24)
project(spin8_prism_graph_test LANGUAGES C CXX)
if(NOT PRISM_SOURCE OR NOT SPIN8_SOURCE)
    message(FATAL_ERROR "Set PRISM_SOURCE and SPIN8_SOURCE")
endif()
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SPIN8_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPIN8_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
add_subdirectory("${PRISM_SOURCE}/ggml" ggml)
add_subdirectory("${SPIN8_SOURCE}" spin8)
add_executable(spin8_prism_graph_test ../spin8_prism_graph_test.cpp)
target_include_directories(spin8_prism_graph_test PRIVATE "${PRISM_SOURCE}/src")
target_link_libraries(spin8_prism_graph_test PRIVATE spin8 ggml ggml-cpu)
target_compile_features(spin8_prism_graph_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(spin8_prism_graph_test PRIVATE /W4 /WX /utf-8)
endif()
