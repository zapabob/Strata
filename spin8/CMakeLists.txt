cmake_minimum_required(VERSION 3.24)
project(spin8 VERSION 0.1.0 LANGUAGES C CXX)
option(SPIN8_ENABLE_CUDA "Build optional CUDA implementation" OFF)
option(SPIN8_BUILD_TESTS "Build independent reference tests" ON)
add_library(spin8 STATIC src/spin8.cpp src/capabilities.cpp src/query.cpp src/query_v3.cpp
 src/query_adapter.cpp third_party/sha256/sha256.c)
target_include_directories(spin8 PUBLIC include)
target_compile_features(spin8 PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(spin8 PRIVATE /W4 /WX /fp:strict)
else()
 target_compile_options(spin8 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SPIN8_BUILD_TESTS)
 enable_testing()
 add_executable(spin8_test tests/cpu_test.cpp)
 target_link_libraries(spin8_test PRIVATE spin8)
 add_test(NAME spin8_cpu COMMAND spin8_test)
 add_executable(spin8_query_test tests/query_test.cpp)
 target_link_libraries(spin8_query_test PRIVATE spin8)
 add_test(NAME spin8_query COMMAND spin8_query_test)
 add_executable(spin8_query_v3_test tests/query_v3_test.cpp)
 target_link_libraries(spin8_query_v3_test PRIVATE spin8)
 add_test(NAME spin8_query_v3 COMMAND spin8_query_v3_test)
 add_executable(spin8_query_adapter_test tests/query_adapter_test.cpp tests/query_adapter_c_test.c)
 target_link_libraries(spin8_query_adapter_test PRIVATE spin8)
 if(MSVC)
  target_compile_options(spin8_query_adapter_test PRIVATE /W4 /WX /fp:strict)
 else()
  target_compile_options(spin8_query_adapter_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME spin8_query_adapter COMMAND spin8_query_adapter_test)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME spin8_query_sidecar_python COMMAND "${Python3_EXECUTABLE}" -B
   "${CMAKE_CURRENT_SOURCE_DIR}/../tools/test_spin8_query_sidecar.py")
  set_tests_properties(spin8_query_sidecar_python PROPERTIES ENVIRONMENT
   "SPIN8_SIDECAR_TEST_EXE=$<TARGET_FILE:spin8_query_adapter_test>;SPIN8_TEST_TMP=${CMAKE_CURRENT_BINARY_DIR}")
 endif()
 add_executable(spin8_c_abi_test tests/c_abi_test.c)
 target_link_libraries(spin8_c_abi_test PRIVATE spin8)
 set_target_properties(spin8_c_abi_test PROPERTIES LINKER_LANGUAGE CXX)
 add_test(NAME spin8_c_abi COMMAND spin8_c_abi_test)
endif()
if(SPIN8_ENABLE_CUDA)
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 if(MSVC AND CMAKE_GENERATOR MATCHES "Visual Studio")
  # CUDA's VS integration otherwise inherits /W1 after our explicit /W4.
  # Match its per-item host warning level without changing numerical flags.
  set_source_files_properties(src/cuda.cu src/query_cuda.cu src/query_v3_cuda.cu
   tests/cuda_test.cu tests/query_cuda_test.cu tests/query_v3_cuda_test.cu
   PROPERTIES VS_SETTINGS "Warning=W4")
 endif()
 add_library(spin8_cuda STATIC src/cuda.cu src/query_cuda.cu src/query_v3_cuda.cu)
 target_include_directories(spin8_cuda PUBLIC include)
 target_link_libraries(spin8_cuda PUBLIC spin8 CUDA::cudart)
 if(MSVC)
  target_compile_options(spin8_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/W4>)
 endif()
 set_target_properties(spin8_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
 if(SPIN8_BUILD_TESTS)
  add_executable(spin8_cuda_test tests/cuda_test.cu)
  target_link_libraries(spin8_cuda_test PRIVATE spin8_cuda)
  if(MSVC)
   target_compile_options(spin8_cuda_test PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/W4>)
  endif()
  set_target_properties(spin8_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_test(NAME spin8_cuda COMMAND spin8_cuda_test)
  add_executable(spin8_query_cuda_test tests/query_cuda_test.cu)
  target_link_libraries(spin8_query_cuda_test PRIVATE spin8_cuda)
  set_target_properties(spin8_query_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  if(MSVC)
   target_compile_options(spin8_query_cuda_test PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/W4>)
  endif()
  add_test(NAME spin8_query_cuda COMMAND spin8_query_cuda_test)
  add_executable(spin8_query_v3_cuda_test tests/query_v3_cuda_test.cu)
  target_link_libraries(spin8_query_v3_cuda_test PRIVATE spin8_cuda)
  set_target_properties(spin8_query_v3_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  if(MSVC)
   target_compile_options(spin8_query_v3_cuda_test PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/W4>)
  endif()
  add_test(NAME spin8_query_v3_cuda COMMAND spin8_query_v3_cuda_test)
  add_executable(spin8_native_adapter_test ../tests/core/spin8_native_adapter_test.cpp ../src/core/spin8_attention.cpp)
  target_include_directories(spin8_native_adapter_test PRIVATE ../include)
  target_link_libraries(spin8_native_adapter_test PRIVATE spin8_cuda)
  target_compile_features(spin8_native_adapter_test PRIVATE cxx_std_20)
  if(MSVC)
   target_compile_options(spin8_native_adapter_test PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME spin8_native_adapter COMMAND spin8_native_adapter_test "${CMAKE_CURRENT_SOURCE_DIR}/../data/spin8/reference.profile")
 endif()
endif()
