cmake_minimum_required(VERSION 3.24)
project(tqcodec LANGUAGES C CXX)
add_library(tqcodec STATIC src/codec.cpp src/cache.cpp src/distill.cpp src/calibrate.cpp)
target_include_directories(tqcodec PUBLIC include)
target_compile_features(tqcodec PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(tqcodec PRIVATE /W4 /WX /fp:strict)
else()
 target_compile_options(tqcodec PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
option(TQCODEC_SPIN8_TESTS "Test strict Spin8 and query-controller composition" OFF)
if(BUILD_TESTING)
 add_executable(tqcodec_c_abi_test tests/c_abi_test.c)
 target_link_libraries(tqcodec_c_abi_test PRIVATE tqcodec)
 set_target_properties(tqcodec_c_abi_test PROPERTIES LINKER_LANGUAGE CXX)
 add_test(NAME tqcodec_c_abi COMMAND tqcodec_c_abi_test)
 add_executable(tqcodec_test tests/codec_test.cpp)
 target_link_libraries(tqcodec_test PRIVATE tqcodec)
 add_test(NAME tqcodec_cpu COMMAND tqcodec_test)
 add_executable(tqcodec_cache_test tests/cache_test.cpp)
 target_link_libraries(tqcodec_cache_test PRIVATE tqcodec)
 add_test(NAME tqcodec_cache COMMAND tqcodec_cache_test)
 add_executable(tqcodec_transport_test tests/transport_test.cpp)
 target_link_libraries(tqcodec_transport_test PRIVATE tqcodec)
 add_test(NAME tqcodec_transport COMMAND tqcodec_transport_test)
 add_executable(tqcodec_distill_test tests/distill_test.cpp)
 target_link_libraries(tqcodec_distill_test PRIVATE tqcodec)
 add_test(NAME tqcodec_distill COMMAND tqcodec_distill_test)
 add_executable(tqcodec_calibrate_test tests/calibrate_test.cpp)
 target_link_libraries(tqcodec_calibrate_test PRIVATE tqcodec)
 add_test(NAME tqcodec_calibrate COMMAND tqcodec_calibrate_test)
 if(TQCODEC_SPIN8_TESTS)
  set(SPIN8_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../spin8 spin8-reference)
  add_executable(tqcodec_spin8_test tests/spin8_integration_test.cpp)
  target_link_libraries(tqcodec_spin8_test PRIVATE tqcodec spin8)
  add_test(NAME tqcodec_spin8 COMMAND tqcodec_spin8_test)
  add_executable(tqcodec_triality_distill_test tests/triality_distill_test.cpp)
  target_link_libraries(tqcodec_triality_distill_test PRIVATE tqcodec spin8)
  add_test(NAME tqcodec_triality_distill COMMAND tqcodec_triality_distill_test)
 endif()
endif()
