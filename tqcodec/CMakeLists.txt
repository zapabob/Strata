cmake_minimum_required(VERSION 3.24)
project(tqcodec LANGUAGES C CXX)
add_library(tqcodec STATIC src/codec.cpp src/cache.cpp src/distill.cpp src/calibrate.cpp)
target_include_directories(tqcodec PUBLIC include)
target_compile_features(tqcodec PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(tqcodec PRIVATE /W4 /WX /fp:strict)
else()
 target_compile_options(tqcodec PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
option(TQCODEC_DIALECTS "Build explicitly versioned pinned CPU F32 dialects" OFF)
if(TQCODEC_DIALECTS)
 target_sources(tqcodec PRIVATE src/dialects.cpp)
 if(NOT MSVC)
  set_source_files_properties(src/dialects.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-fno-fast-math")
 endif()
endif()
option(TQCODEC_SPIN8_TESTS "Test strict Spin8 and query-controller composition" OFF)
set(TQCODEC_ORACLE_FIXTURES "" CACHE PATH "Independent pinned oracle fixture directory")
option(TQCODEC_DIALECTS_CUDA "Build exact asynchronous TheTom Turbo KV CUDA dialects" OFF)
if(TQCODEC_DIALECTS_CUDA)
 enable_language(CUDA)
 find_package(CUDAToolkit 12.8 REQUIRED)
 add_library(tqcodec_dialects_cuda STATIC src/dialects_cuda.cu)
 target_include_directories(tqcodec_dialects_cuda PUBLIC include)
 target_compile_features(tqcodec_dialects_cuda PUBLIC cxx_std_17)
 set_target_properties(tqcodec_dialects_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 120)
 target_compile_options(tqcodec_dialects_cuda PRIVATE
  $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true>)
 target_link_libraries(tqcodec_dialects_cuda PUBLIC CUDA::cudart)
 if(MSVC)
  target_compile_options(tqcodec_dialects_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/W4>)
  if(CMAKE_GENERATOR MATCHES "Visual Studio")
   set_source_files_properties(src/dialects_cuda.cu PROPERTIES VS_SETTINGS "Warning=W4")
  endif()
 endif()
endif()
if(BUILD_TESTING)
 # Compile the public C header even in a build with no CUDA language/toolkit.
 add_library(tqcodec_dialects_cuda_header OBJECT tests/dialects_cuda_c_test.c)
 target_include_directories(tqcodec_dialects_cuda_header PRIVATE include)
 if(MSVC)
  target_compile_options(tqcodec_dialects_cuda_header PRIVATE /W4 /WX /utf-8)
 endif()
 if(TQCODEC_DIALECTS_CUDA)
  add_executable(tqcodec_dialects_cuda_c_test tests/dialects_cuda_c_test.c)
  set_target_properties(tqcodec_dialects_cuda_c_test PROPERTIES LINKER_LANGUAGE CXX)
  target_link_libraries(tqcodec_dialects_cuda_c_test PRIVATE tqcodec_dialects_cuda)
  add_test(NAME tqcodec_dialects_cuda_c COMMAND tqcodec_dialects_cuda_c_test)
  if(TQCODEC_DIALECTS)
   add_executable(tqcodec_dialects_cuda_test tests/dialects_cuda_test.cpp)
   target_link_libraries(tqcodec_dialects_cuda_test PRIVATE tqcodec_dialects_cuda tqcodec)
   if(MSVC)
    target_compile_options(tqcodec_dialects_cuda_test PRIVATE /W4 /WX /fp:strict /utf-8)
   else()
    target_compile_options(tqcodec_dialects_cuda_test PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
   endif()
   add_test(NAME tqcodec_dialects_cuda COMMAND tqcodec_dialects_cuda_test
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/dialects-v1.bin")
   if(TQCODEC_ORACLE_FIXTURES)
    add_test(NAME tqcodec_dialects_cuda_oracle COMMAND tqcodec_dialects_cuda_test
     "${TQCODEC_ORACLE_FIXTURES}/thetom-fixtures.bin")
   endif()
  endif()
 endif()
 if(TQCODEC_DIALECTS)
  add_executable(tqcodec_dialects_test tests/dialects_test.cpp)
  target_link_libraries(tqcodec_dialects_test PRIVATE tqcodec)
  add_test(NAME tqcodec_dialects COMMAND tqcodec_dialects_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/dialects-v1.bin")
  add_executable(tqcodec_dialects_c_test tests/dialects_c_test.c)
  target_link_libraries(tqcodec_dialects_c_test PRIVATE tqcodec)
  set_target_properties(tqcodec_dialects_c_test PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME tqcodec_dialects_c COMMAND tqcodec_dialects_c_test)
  if(TQCODEC_ORACLE_FIXTURES)
   add_test(NAME tqcodec_dialects_oracle COMMAND tqcodec_dialects_test
    "${TQCODEC_ORACLE_FIXTURES}/thetom-fixtures.bin"
    "${TQCODEC_ORACLE_FIXTURES}/prism-fixtures.bin"
    "${TQCODEC_ORACLE_FIXTURES}/native-fixtures.bin")
  endif()
  if(MSVC)
   target_compile_options(tqcodec_dialects_test PRIVATE /W4 /WX /fp:strict)
   target_compile_options(tqcodec_dialects_c_test PRIVATE /W4 /WX /fp:strict)
  else()
   target_compile_options(tqcodec_dialects_test PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
  endif()
 endif()
 add_executable(tqcodec_c_abi_test tests/c_abi_test.c)
 target_link_libraries(tqcodec_c_abi_test PRIVATE tqcodec)
 set_target_properties(tqcodec_c_abi_test PROPERTIES LINKER_LANGUAGE CXX)
 add_test(NAME tqcodec_c_abi COMMAND tqcodec_c_abi_test)
 add_executable(tqcodec_test tests/codec_test.cpp)
 target_link_libraries(tqcodec_test PRIVATE tqcodec)
 add_test(NAME tqcodec_cpu COMMAND tqcodec_test)
 add_executable(tqcodec_cache_test tests/cache_test.cpp)
 target_link_libraries(tqcodec_cache_test PRIVATE tqcodec)
 add_test(NAME tqcodec_cache COMMAND tqcodec_cache_test)
 add_executable(tqcodec_transport_test tests/transport_test.cpp)
 target_link_libraries(tqcodec_transport_test PRIVATE tqcodec)
 add_test(NAME tqcodec_transport COMMAND tqcodec_transport_test)
 add_executable(tqcodec_distill_test tests/distill_test.cpp)
 target_link_libraries(tqcodec_distill_test PRIVATE tqcodec)
 add_test(NAME tqcodec_distill COMMAND tqcodec_distill_test)
 add_executable(tqcodec_calibrate_test tests/calibrate_test.cpp)
 target_link_libraries(tqcodec_calibrate_test PRIVATE tqcodec)
 add_test(NAME tqcodec_calibrate COMMAND tqcodec_calibrate_test)
 if(TQCODEC_SPIN8_TESTS)
  set(SPIN8_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../spin8 spin8-reference)
  add_executable(tqcodec_spin8_test tests/spin8_integration_test.cpp)
  target_link_libraries(tqcodec_spin8_test PRIVATE tqcodec spin8)
  add_test(NAME tqcodec_spin8 COMMAND tqcodec_spin8_test)
  add_executable(tqcodec_triality_distill_test tests/triality_distill_test.cpp)
  target_link_libraries(tqcodec_triality_distill_test PRIVATE tqcodec spin8)
  add_test(NAME tqcodec_triality_distill COMMAND tqcodec_triality_distill_test)
 endif()
endif()
option(TQCODEC_DEVICE_CACHE "Build opt-in sm120 device cache" OFF)
if(TQCODEC_DEVICE_CACHE)
 if(NOT TQCODEC_DIALECTS_CUDA OR NOT TQCODEC_DIALECTS)
  message(FATAL_ERROR "Device cache requires explicit CPU and CUDA dialects")
 endif()
 if(NOT TARGET spin8)
  set(SPIN8_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../spin8 spin8-cache)
 endif()
 add_library(tqcodec_device_cache STATIC src/cache_cuda_plan.cpp src/cache_cuda.cu src/cache_cuda_snapshot.cpp)
 target_include_directories(tqcodec_device_cache PUBLIC include)
 target_link_libraries(tqcodec_device_cache PUBLIC tqcodec_dialects_cuda tqcodec spin8)
 target_compile_features(tqcodec_device_cache PUBLIC cxx_std_17)
 set_target_properties(tqcodec_device_cache PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 120)
 target_compile_options(tqcodec_device_cache PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true>)
 if(MSVC)
  target_compile_options(tqcodec_device_cache PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/fp:strict;/utf-8> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/W4>)
  set_source_files_properties(src/cache_cuda.cu PROPERTIES VS_SETTINGS "Warning=W4")
 endif()
 if(BUILD_TESTING)
  add_executable(tqcodec_cache_cuda_test tests/cache_cuda_test.cpp)
  target_link_libraries(tqcodec_cache_cuda_test PRIVATE tqcodec_device_cache)
  if(MSVC)
   target_compile_options(tqcodec_cache_cuda_test PRIVATE /W4 /WX /fp:strict /utf-8)
  endif()
  add_test(NAME tqcodec_cache_cuda COMMAND tqcodec_cache_cuda_test)
 endif()
 if(BUILD_TESTING)
  add_executable(tqcodec_cache_cuda_c_test tests/cache_cuda_c_test.c)
  set_target_properties(tqcodec_cache_cuda_c_test PROPERTIES LINKER_LANGUAGE CXX)
  target_link_libraries(tqcodec_cache_cuda_c_test PRIVATE tqcodec_device_cache)
  add_test(NAME tqcodec_cache_cuda_c COMMAND tqcodec_cache_cuda_c_test)
 endif()
endif()
if(BUILD_TESTING)
 add_library(tqcodec_cache_cuda_header OBJECT tests/cache_cuda_c_test.c)
 target_include_directories(tqcodec_cache_cuda_header PRIVATE include)
endif()
